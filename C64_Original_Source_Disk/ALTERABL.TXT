; put "alterables"
;
; choose k/j & q/s
;
; default values are :-
;
; keyboard    joysel = 0
;
; sound on    sndflg = 1
;
alter
;
jsr inkey
cmp #3       ;* 'c'
bne alt1     ;* not pressed
;
lda joysel
eor #1
sta joysel
;
ldx #47
;
alt2
;
lda $2440,x  ;* existing char
pha
;
lda jkchar,x ;* new one
sta $2440,x
;
pla
sta jkchar,x ;* replace it
;
dex
bpl alt2
;
;
alt1
;
jsr inkey
cmp #19      ;* 's'
bne alt3
;
lda sndflg
eor #1
sta sndflg
;
ldx #47
;
alt4
;
lda $2470,x
pha
;
lda sqchar,x
sta $2470,x
;
pla
sta sqchar,x
;
dex
bpl alt4
;
alt3
;
rts
